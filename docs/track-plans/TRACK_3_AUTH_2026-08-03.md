## STEP 1 — Create `components/shared/AuthGate.tsx`

Extract the `App.tsx:90-148` bootstrap into a reusable component that wraps children. It performs the same 3 jobs: session check, portal guard, auth-state subscription.

```tsx
import React, { useEffect, useState } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import { getCurrentUser, hasAccessToPortal } from '../../services/AuthService';
import { supabase } from '../../services/supabaseClient';
import { useAppStore } from '../../store/useAppStore';
import { PageLoader } from './PageLoader';   // verify this exists; App.tsx uses a loader fallback

interface AuthGateProps {
  portal: 'admin' | 'teacher' | 'student' | 'parent';
  children: React.ReactNode;
}

const PUBLIC_PATHS = ['/login', '/', '/claim'];   // mirror App.tsx:109-114

export const AuthGate: React.FC<AuthGateProps> = ({ portal, children }) => {
  const { setUserProfile, clearUserProfile } = useAppStore();
  const navigate = useNavigate();
  const location = useLocation();
  const [ready, setReady] = useState(false);

  // Session bootstrap
  useEffect(() => {
    let mounted = true;
    (async () => {
      try {
        const user = await getCurrentUser();
        if (!mounted) return;
        if (user) {
          setUserProfile(user);
          if (!hasAccessToPortal(user.role, portal)) {
            // role can't access this portal — send them to their home
            const home = homePathForRole(user.role);
            navigate(home, { replace: true });
            return;
          }
        } else {
          clearUserProfile();
          // not on a public path → go to login (the SPA-level /login lives under the main entry,
          // so for standalone entries we redirect to the root login)
          if (!isPublicPath(location.pathname)) {
            window.location.href = '/login';
            return;
          }
        }
        setReady(true);
      } catch {
        if (!mounted) return;
        clearUserProfile();
        if (!isPublicPath(location.pathname)) {
          window.location.href = '/login';
          return;
        }
        setReady(true);
      }
    })();
    return () => { mounted = false; };
  }, [portal, setUserProfile, clearUserProfile, navigate, location.pathname]);

  // Auth-state subscription (mirror App.tsx:130-144)
  useEffect(() => {
    const { data: sub } = supabase.auth.onAuthStateChange((event) => {
      if (event === 'SIGNED_OUT') {
        clearUserProfile();
        window.location.href = '/login';
      } else if (event === 'USER_UPDATED' || event === 'TOKEN_REFRESHED') {
        getCurrentUser().then(u => u ? setUserProfile(u) : clearUserProfile());
      }
    });
    return () => sub.subscription.unsubscribe();
  }, [setUserProfile, clearUserProfile]);

  if (!ready) return <PageLoader />;   // or a centered spinner
  return <>{children}</>;
};

function isPublicPath(p: string): boolean {
  return PUBLIC_PATHS.some(pp => p === pp || p.startsWith(pp));
}
function homePathForRole(role: string): string {
  if (role === 'admin' || role === 'manager') return '/admin';
  if (role === 'teacher') return '/teacher';
  if (role === 'parent') return '/parent';
  return '/student';
}
```

**Key design decisions:**
- **`window.location.href = '/login'`** for the no-user case (not `navigate('/login')`) — because the standalone entries have `basename="/admin"` etc., an in-router navigate would go to `/admin/login`. The shared login UI lives at the root `/login` served by `index.html`. Verify this matches how `App.tsx` handles it (it uses `navigate('/login')` because it's on the root router). If standalone entries need an in-portal login screen, that's a larger change — **default to redirecting to root `/login`** for v1.
- **`PageLoader`** — confirm the component name (audit referenced it but didn't confirm the path). If absent, use a simple centered spinner.
- **The `hasAccessToPortal` redirect** means a student who manually visits `/admin` gets bounced to `/student`. Good.

## STEP 2 — Wire AuthGate into the 4 entries

For each `*Entry.tsx`, wrap the routed app in `<AuthGate portal="...">`. Example for `adminEntry.tsx`:

```tsx
// adminEntry.tsx
<ErrorBoundary>
  <AppProviders>
    <MockModeBanner />
    <UpdatePrompt />
    <SessionProvider>
      <Toaster />
      <BrowserRouter basename="/admin">
        <AuthGate portal="admin">
          <AdminEntry />     // ← AdminEntry stays; its internal route renders <AdminPortal/>
        </AuthGate>
      </BrowserRouter>
    </SessionProvider>
  </AppProviders>
</ErrorBoundary>
```

**Critical `adminEntry` fix:** `AdminEntry` (the inner component, lines 23-43) currently lazy-imports and renders `DistrictAdminDashboard` directly. Change it to render `<AdminPortal/>` instead — that's the component that does the `userProfile?.role === 'manager' ? <ManagerDashboard/> : <DistrictAdminDashboard/>` split (verified at `AdminPortal.tsx:22`). Without this, even with auth, managers see the District view.

Repeat the `<AuthGate>` wrap for `teacherEntry.tsx` (portal="teacher"), `studentEntry.tsx` (portal="student", note it uses `SoloSessionProvider`), `parentEntry.tsx` (portal="parent").

**One subtlety for `teacherEntry`:** it has a `/live` route (lines 44-46). AuthGate wraps the whole router, so `/live` is still gated — correct (a teacher must be signed in to run a live session).
